(* Simple tree evaluator *)

datatype TreeOp = Add | Sub | Mul | Div | Pow
datatype Tree = Leaf of int
              | Node of Tree * TreeOp * Tree

fun intPow n p =
  Real.round (Math.pow (Real.fromInt n, Real.fromInt p))

fun eval (Leaf n) = n
  | eval (Node(lhs, Add, rhs)) = (eval lhs) + (eval rhs)
  | eval (Node(lhs, Sub, rhs)) = (eval lhs) - (eval rhs)
  | eval (Node(lhs, Mul, rhs)) = (eval lhs) * (eval rhs)
  | eval (Node(lhs, Div, rhs)) = (eval lhs) div (eval rhs)
  | eval (Node(lhs, Pow, rhs)) = intPow (eval lhs) (eval rhs)

fun assert false = raise Fail "assertion failure"
  | assert true = true

;

let
  (* 5 + 2 * 3 *)
  val n1 = eval (Node(Leaf 5, Add, Node(Leaf 2, Mul, Leaf 3)))
  val n2 = eval (Node(Leaf 3, Mul, Leaf 3))
  (* 13 * 3 + 48 / (8 - 4) *)
  val n3 = eval (Node(
                   Node(Leaf 13, Mul, Leaf 3),
                   Add,
                   Node(Leaf 48, Div, Node(Leaf 8, Sub, Leaf 4))))
  val n4 = eval (Leaf 39)
  (* 2^8 * 5^4 *)
  val n5 = eval (Node(Node(Leaf 2, Pow, Leaf 8), Add, Node(Leaf 5, Pow, Leaf 4)))
in
  assert (n1 = 11);
  assert (n2 = 9);
  assert (n3 = 51);
  assert (n4 = 39);
  assert (n5 = 881)
end
